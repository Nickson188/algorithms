using ull = unsigned long long;
const int N=1000010;
const int B=131;
ull h[N],b[N]; //h[i]存s前缀哈希值，b[i]存B的i次幂
ull get(int l,int r){ //区间哈希值
  return h[r]-h[l-1]*b[r-l+1]; //h[r]-h[l-1]*B^(r-l+1)
}
void pre(string &s)
{
    int n=s.size();
    s=" "+s;
    h[0]=0;
    b[0]=1;
    for(int i=1;i<=n;i++)
    {
        h[i]=h[i-1]*B+s[i]; //h[abcde]=a*B^4+b*B^3+c*B^2+d*B^1+e
        b[i]=b[i-1]*B; //b[i]=B^i
    }
}
bool query(int a,int b,int c,int d)
{
    if(get(a,b)==get(c,d)) 
    {
        return true;
    }
    else
    {
        return false;
    }
}

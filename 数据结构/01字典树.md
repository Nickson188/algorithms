// 给定 N 个整数，选出两个进行异或计算，求最大异或和
const int N = 100010;
int a[N];
int ch[N*31][2];
int idx;
int sum[N];
void insert(int x)
{
    int p=0;
    for(int i=30;i>=0;i--)
    {
        int j=x>>i&1;
        if(!ch[p][j])
        {
            ch[p][j]=++idx;
        }
        p=ch[p][j];
    }
}
int query(int x)
{
    int p=0;
    int res=0;
    for(int i=30;i>=0;i--)
    {
        int j=x>>i&1;
        if(ch[p][!j])
        {
            res+=1LL<<i;
            p=ch[p][!j];
        }
        else
        {
            p=ch[p][j];
        }
    }
    return res;
}

```cpp
// 向量u=(ux,uy)
// 向量v=(vx,vy)
int ux,uy;
int vx,vy;
// 叉积
int cross=ux*vy-uy*vx;
// cross>0 向量v在向量u的逆时针方向
// cross<0 向量v在向量u的顺时针方向
// cross=0 向量v在向量u平行(同向或反向)

struct point
{
    int x;
    int y;
};
int cross(const point&i,const point&j,const point&k)
{
    int ijx=j.x-i.x;
    int ijy=j.y-i.y;
    int ikx=k.x-i.x;
    int iky=k.y-i.y;
    int cs=ijx*iky-ijy*ikx;
    return cs;
}
// 用于判断向量ik是在向量ij什么方向
```

// 假设 n 为字符串长度
vector<long long> d(n + 2, 0);
vector<long long> dd(n + 2, 0);

// --- 每次区间修改的操作 ---
// 例如：从 l 开始，初始贡献为 cnt，之后每往右一步 -1，持续到 r 结束
auto change = [&](int l, int r, long long cnt) {
    if (l > r) return;
    
    // 1. 在左端点 l 加上初始值
    d[l] += cnt;
    // 2. 在 l+1 处开始施加斜率（公差为 -1 的趋势）
    dd[l + 1]--;
    
    // 3. 在 r+1 处结束这个趋势（恢复平稳）
    if (r + 1 < n) {
        d[r + 1] -= cnt - (r - l + 1);
        dd[r + 1]++;
    }
};

// --- 最后的前缀和还原 ---
vector<long long> ans(n, 0);
long long sum = 0, add = 0;
for (int i = 0; i < n; ++i) {
    sum += d[i];
    add += dd[i];
    sum += add;
    ans[i] = sum;
}

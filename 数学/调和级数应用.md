```cpp
// 计算 1 到 N 每个数的所有因数和
vector<long long> divisorSum(N + 1, 0);

for (int i = 1; i <= N; i++) {
    for (int j = i; j <= N; j += i) {
        divisorSum[j] += i;
    }
}
// 复杂度：O(N log N)



// 统计数组中每个数的倍数出现次数
vector<int> cnt(N + 1);  // cnt[x] = x 在数组中出现的次数
vector<int> multiples(N + 1);

for (int i = 1; i <= N; i++) {
    for (int j = i; j <= N; j += i) {
        multiples[i] += cnt[j];  // i 的倍数个数
    }
}
// 复杂度：O(N log N)



// 预处理 1 到 N 的所有因子
vector<vector<int>> factors(N + 1);

for (int i = 1; i <= N; i++) {
    for (int j = i; j <= N; j += i) {
        factors[j].push_back(i);
    }
}
// 复杂度：O(N log N)
// 空间：O(N log N)（每个数平均有 log N 个因子）
```

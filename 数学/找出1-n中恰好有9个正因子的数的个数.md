// 如果一个数分解质因数为 M=p1^a1*p2^a2*...*pk^ak
// p1 p2 pk是不同的质数,a1 a2 ak是指数
// 则 M 的约数为 d(M)=(a1+1)*(a2+1)*...*(ak+1)
// M=36=2^2*3^3
// 2的指数:0,1,2
// 3的指数:0,1,2
// 所有组合：总共=(2+1)*(2+1)=9
// 2⁰×3⁰=1
// 2⁰×3¹=3
// 2⁰×3²=9
// 2¹×3⁰=2
// 2¹×3¹=6
// 2¹×3²=18
// 2²×3⁰=4
// 2²×3¹=12
// 2²×3²=36
#include <bits/stdc++.h>
using namespace std;
#define int long long

int dx[4] = {0, 0, 1, -1};

const int P = 998244353;

//欧拉筛
std::vector<int> minp, primes;
// minp[i] i的最小质因子
void sieve(int n) {
    minp.assign(n + 1, 0);primes.clear();

    for (int i = 2; i <= n; i++) {
        if (minp[i] == 0) {
            minp[i] = i;
            primes.push_back(i);
        }

        for (auto p : primes) {
            if (i * p > n) break;
            minp[i * p] = p;
            
            if (p == minp[i]) break;
        }
    }
}
signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T = 1;
    // cin >> T;
    while(T--)
    {
        int n;
        cin>>n;
        int sz=sqrt(n);
        sieve(sz+10);
        int ans=0;
        // p^8
        // (1+8)*(0+1)=9
        for(auto &x:primes)
        {
            int temp=1;
            int flag=0;
            for(int i=0;i<8;i++)
            {
                if(temp>n/x)
                {
                    flag=1;
                    break;
                }
                temp*=x;
            }
            if(!flag)
            {
                ans++;
            }
        }
        // p^2*q^2
        // (2+1)*(2+1)=9
        int m=primes.size();
        int qidx=m-1;
        for(int i=0;i<m;i++)
        {
            int p=primes[i];
            int p2=p*p;
            if(p2>n/p2) break;
            while(qidx>i and p2>n/(primes[qidx]*primes[qidx]))
            {
                qidx--;
            }
            if(qidx>i)
            {
                ans+=qidx-i;
            }
        }
        cout<<ans<<endl;
    }
    return 0; 
}

const int N=100010;
const int INF=1e18;
int w[N];
int fa[N],dep[N],siz[N],son[N];
int top[N],dfn[N],rnk[N];
// fa[u] 存u的父节点 dep[u] 存u的深度
// son[u] 存u的重儿子 sz[u] 存以u为根的子树节点数
// top[u] 存u所在重链的顶点 dfn[u] 存u剖分后的新编号
// rnk[cnt] 存新编号在树中所对应节点的权值
int n,m,R,P;
struct SGT {
    #define lc u << 1
    #define rc u << 1 | 1
    struct Node {
        int sum;
        int minv;
        int maxv;
        int setv;
        int addv;
    } tr[N * 4];

    void pushup(int u) {
        tr[u].sum = (tr[lc].sum%P + tr[rc].sum%P)%P;
        tr[u].minv = min(tr[lc].minv%P, tr[rc].minv%P);
        tr[u].maxv = max(tr[lc].maxv%P, tr[rc].maxv%P);
    }

    void apply_set(int u, int len, int val)
    {
        val%=P;
        tr[u].setv = val;
        tr[u].addv = 0;
        tr[u].sum = val * len;
        tr[u].minv = val;
        tr[u].maxv = val;
    }

    void apply_add(int u, int len, int val)
    {
        val%=P;
        if (tr[u].setv != -1)
        {
            tr[u].setv += val;
            tr[u].sum = tr[u].setv * len;
            tr[u].minv = tr[u].setv;
            tr[u].maxv = tr[u].setv;
        }
        else
        {
            tr[u].addv += val;
            tr[u].sum += val * len;
            tr[u].minv += val;
            tr[u].maxv += val;
        }
    }

    void pushdown(int u, int l, int r) {
        int mid = (l + r) >> 1;
        int left_len = mid - l + 1;
        int right_len = r - mid;
        if (tr[u].setv != -1)
        {
            apply_set(lc, left_len, tr[u].setv);
            apply_set(rc, right_len, tr[u].setv);
            tr[u].setv = -1;
        }
        if (tr[u].addv != 0)
        {
            apply_add(lc, left_len, tr[u].addv);
            apply_add(rc, right_len, tr[u].addv);
            tr[u].addv = 0;
        }
    }

    void build(int u, int l, int r) {
        tr[u].setv = -1;
        tr[u].addv = 0;
        if (l == r) {
            tr[u].sum = w[l];
            tr[u].minv = w[l];
            tr[u].maxv = w[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(lc, l, mid);
        build(rc, mid + 1, r);
        pushup(u);
    }

    void update_set(int u, int l, int r, int x, int y, int val) {
        val%=P;
        if (x > r || y < l) return;
        if (x <= l && r <= y) {
            apply_set(u, r - l + 1, val);
            return;
        }
        pushdown(u, l, r);
        int mid = (l + r) >> 1;
        update_set(lc, l, mid, x, y, val);
        update_set(rc, mid + 1, r, x, y, val);
        pushup(u);
    }

    void update_add(int u, int l, int r, int x, int y, int val) {
        val%=P;
        if (x > r || y < l) return;
        if (x <= l && r <= y) {
            apply_add(u, r - l + 1, val);
            return;
        }
        pushdown(u, l, r);
        int mid = (l + r) >> 1;
        update_add(lc, l, mid, x, y, val);
        update_add(rc, mid + 1, r, x, y, val);
        pushup(u);
    }

    int query_min(int u, int l, int r, int x, int y) {
        if (x > r || y < l) return INF;
        if (x <= l && r <= y) return tr[u].minv%P;
        pushdown(u, l, r);
        int mid = (l + r) >> 1;
        return min(query_min(lc, l, mid, x, y)%P, query_min(rc, mid + 1, r, x, y)%P);
    }

    int query_max(int u, int l, int r, int x, int y) {
        if (x > r || y < l) return -INF;
        if (x <= l && r <= y) return tr[u].maxv%P;
        pushdown(u, l, r);
        int mid = (l + r) >> 1;
        return max(query_max(lc, l, mid, x, y)%P, query_max(rc, mid + 1, r, x, y)%P);
    }

    int query_sum(int u, int l, int r, int x, int y) {
        if (x > r || y < l) return 0;
        if (x <= l && r <= y) return tr[u].sum%P;
        pushdown(u, l, r);
        int mid = (l + r) >> 1;
        return (query_sum(lc, l, mid, x, y)%P + query_sum(rc, mid + 1, r, x, y)%P)%P;
    }
} tree;
// 树链剖分
vector<int> e[N];
void dfs1(int u,int f){ //搞fa,dep,siz,son
  fa[u]=f,dep[u]=dep[f]+1,siz[u]=1;
  for(int v:e[u])if(v!=f){
    dfs1(v,u);
    siz[u]+=siz[v];
    if(siz[son[u]]<siz[v]) son[u]=v; 
  }
}
void dfs2(int u,int t){ //搞top,dfn,rnk
  top[u]=t,dfn[u]=++dfn[0],rnk[dfn[0]]=u;
  if(son[u]) dfs2(son[u],t);
  for(int v:e[u])if(v!=fa[u]&&v!=son[u]) dfs2(v,v);
}
// 从节点u到v最短路径所有节点值都加上k
void add_path(int u,int v,int k)
{
    k%=P;
    while(top[u]!=top[v])
    {
        if(dep[top[u]]<dep[top[v]])
        {
            swap(u,v);
        }
        tree.update_add(1,1,n,dfn[top[u]],dfn[u],k);
        u=fa[top[u]];
    }
    if(dep[u]>dep[v])
    {
        swap(u,v);
    }
    tree.update_add(1,1,n,dfn[u],dfn[v],k);
}
// 求节点u到v最短路径所有节点值的和
int query_path(int u,int v)
{
    int res=0;
    while(top[u]!=top[v])
    {
        if(dep[top[u]]<dep[top[v]])
        {
            swap(u,v);
        }
        res=(res+tree.query_sum(1,1,n,dfn[top[u]],dfn[u]))%P;
        u=fa[top[u]];
    }
    if(dep[u]>dep[v])
    {
        swap(u,v);
    }
    res=(res+tree.query_sum(1,1,n,dfn[u],dfn[v]))%P;
    return res;
}
// 将以u为跟节点的子树内所有节点都加上k
void add_subtree(int u,int k)
{
    k%=P;
    tree.update_add(1,1,n,dfn[u],dfn[u]+siz[u]-1,k);
    // 子树在DFS序中是连续的一段
}
// 求以u为根节点的子树内所有节点值之和
int query_subtree(int u)
{
    return tree.query_sum(1,1,n,dfn[u],dfn[u]+siz[u]-1);
    // 子树在DFS序中是连续的一段
}
